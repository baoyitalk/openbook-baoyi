# 防抖与节流函数的区别


## 案例

防抖函数与节流函数对比
```js
// 防抖：最后一次
function debounce(cb, delay) {
  let timer;
  return function(...args) {
    clearTimeout(timer);              // 每次都清
    timer = setTimeout(() => cb.apply(this, args), delay);
  };
}

// 节流：固定频率
function throttle(cb, delay) {
  let timer;
  return function(...args) {
    if (timer) return;                // 有定时器就不执行
    timer = setTimeout(() => {
      cb.apply(this, args);
      timer = null;                   // 执行后清空
    }, delay);
  };
}

```

## this指向需要保存吗

搞清楚 外层函数即主函数， 内层函数即回调函数，外层函数很明显是button调用依然不变
但是 主函数内部调用的回调函数

```js
// ❌ 错误写法
function debounce(cb, delay) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => {
      cb(...args);  // ← 这里的 this 是 undefined！
    }, delay);
  };
}

// 使用
button.addEventListener('click', debounce(function() {
  console.log(this);  // undefined，不是 button！
}, 300));

```

```js

// ✅ 正确写法
function debounce(cb, delay) {
  let timer;
  // ...args默认是event数组收集浏览器事件如果这里只打印this也可以不用
  // 作为通用防抖还是要加上...args可以理解为已知的浏览器参数
  return function(...args) {
    const context = this;  // 保存当前的 this
    clearTimeout(timer);
    timer = setTimeout(() => {
      cb.apply(context, args);  // 用 apply 传递正确的 this
    }, delay);
  };
}
// 使用
button.addEventListener('click', debounce(function() {
  console.log(this);  // button！
}, 300));
```

# 写一个防抖组件



```js
import { useCallback, useRef } from 'react';

function useDebounce(callback, delay) {
  const timerRef = useRef(null);
  
  const debouncedCallback = useCallback((...args) => {
    // 清除上一次的定时器
    if (timerRef.current) {
      clearTimeout(timerRef.current);
    }
    
    // 设置新的定时器
    timerRef.current = setTimeout(() => {
      callback(...args);
    }, delay);
  }, [callback, delay]);
  
  return debouncedCallback;
}

export default useDebounce;

```

function SearchBox() {
  const [keyword, setKeyword] = useState('');
  const [results, setResults] = useState([]);
  
  // 搜索函数
  const search = async (value) => {

    const data = await res.json();
    setResults(data);

        value={keyword}
        onChange={(e) => {
          setKeyword(e.target.value);
          debouncedSearch(e.target.value);
        }}
        placeholder="搜索..."
      />
      <ul>
        {results.map(item => (
          <li key={item.id}>{item.name}</li>
        ))}
      </ul>
    </div>
  );
