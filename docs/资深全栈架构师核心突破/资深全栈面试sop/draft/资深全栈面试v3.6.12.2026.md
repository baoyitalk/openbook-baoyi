    where: { id: from },
    data: { balance: { decrement: amount } }

  });

// 2. 没有幂等性控制（重复请求会重复扣款）
// 3. 没有余额检查（可能透支）

```typescript
// ✅ 人工审查后的代码（金融级）
async function transferMoney(
  from: string, 
  to: string, 
  amount: number,
  idempotencyKey: string // 幂等性key
) {
  // 1. 检查幂等性
  const existing = await db.transaction.findUnique({
    where: { idempotencyKey }
  });
  if (existing) {
    return existing; // 重复请求直接返回
  }
  
  // 2. 使用数据库事务
  return await db.$transaction(async (tx) => {
    // 3. 行级锁 + 余额检查
    const fromAccount = await tx.account.findUnique({
      where: { id: from },
      select: { balance: true }
    });
    
    if (!fromAccount || fromAccount.balance < amount) {
      throw new Error('余额不足');
    }
    
    // 4. 原子性扣款和入账
    await tx.account.update({
      where: { id: from },
      data: { balance: { decrement: amount } }
    });
    
    await tx.account.update({
      where: { id: to },
      data: { balance: { increment: amount } }
    });
    
    // 5. 记录交易流水
    return await tx.transaction.create({
      data: {
        idempotencyKey,
        from,
        to,
        amount,
        status: 'SUCCESS'
      }
    });
  });
}
```

## 🚨 2026年必考的"超级Bug"

### Bug 1：Hydration Error（Next.js SSR致命问题）

**第一性原理**：
SSR是服务端生成HTML，客户端再用JS接管。如果服务器和客户端渲染的HTML不一致，React会崩溃。

**典型场景**：
```typescript
// ❌ 错误：服务端和客户端时间不一致
function TimeDisplay() {
  return <div>{new Date().toLocaleString()}</div>;
}

// 服务器渲染：2026-03-06 15:30:00
// 客户端渲染：2026-03-06 15:30:05
// 结果：Hydration Error
```

**资深解法A：两步渲染**
```typescript
function TimeDisplay() {
  const [time, setTime] = useState<string | null>(null);
  
  useEffect(() => {
    setTime(new Date().toLocaleString());
  }, []);
  
  if (!time) return <Skeleton />; // 服务端和首屏都渲染占位符
  return <div>{time}</div>; // 只有客户端渲染真实时间
}
```

**资深解法B：禁用SSR**
```typescript
// 对于重度依赖浏览器的组件（如图表），直接不让它在服务端跑
const NoSSRChart = dynamic(() => import('./Chart'), { ssr: false });
```

**面试话术**：
> "在OmniSettlement的实时看板中，我遇到过Hydration Error。因为服务端渲染的价格和客户端WebSocket推送的价格不一致。我的解决方案是：首屏用骨架屏占位，等客户端挂载后再显示实时数据。这样既保证了SSR的SEO优势，又避免了水合错误。"

### Bug 2：JavaScript精度问题（金融场景致命）

**第一性原理**：
JavaScript只有64位浮点数，无法精确表示某些小数。

```javascript
0.1 + 0.2 === 0.3 // false
0.1 + 0.2 // 0.30000000000000004
```

**金融场景的灾难**：
```typescript
// ❌ 错误：用Number存储金额
const balance = 100.1;
const amount = 0.2;
const newBalance = balance - amount; // 99.89999999999999

// 如果这是用户余额，会导致对账失败
```

**资深解法**：
```typescript
// ✅ 正确：用BigInt或Decimal.js
import Decimal from 'decimal.js';

const balance = new Decimal('100.1');
const amount = new Decimal('0.2');
const newBalance = balance.minus(amount); // 99.9（精确）

// 或者：用分为单位存储
const balanceInCents = 10010; // 100.10元 = 10010分
const amountInCents = 20;     // 0.20元 = 20分
const newBalanceInCents = balanceInCents - amountInCents; // 9990分 = 99.90元
```

**面试话术**：
> "在OmniSettlement中，我规定所有涉及金额的计算必须使用Decimal.js或BigInt，绝对禁止直接使用Number。并且在后端Java层进行二次精度校验。这是金融系统的基本素养。"

### Bug 3：Next.js 15的异步Params陷阱

**第一性原理**：
Next.js 15将路由参数改为异步，这是一个破坏性变更。

```typescript
// ❌ Next.js 14的写法（在15中会crash）
export default function Page({ params }: { params: { id: string } }) {
  return <div>ID: {params.id}</div>;
}

// ✅ Next.js 15的正确写法
export default async function Page({ 
  params 
}: { 
  params: Promise<{ id: string }> 
}) {
  const { id } = await params; // 必须await
  return <div>ID: {id}</div>;
}
```

**面试话术**：
> "这反映了我对最新技术的敏锐度。很多开发者升级到Next.js 15后，代码直接crash，就是因为没注意到params变成了Promise。我在凌动项目升级时，第一时间就发现了这个变更，并在团队内部做了技术分享。"

## 📋 第一性原理：Java vs JavaScript 底层对比表

### 1. 并发模型对比（核心中的核心）

| 维度 | Java（多线程同步） | JavaScript（单线程异步） | 面试必答点 |
|------|-------------------|------------------------|-----------|
| **执行模型** | 多线程并行执行 | 单线程 + Event Loop | "JS是单线程，但通过异步I/O实现并发" |
| **阻塞处理** | 线程阻塞，其他线程继续 | 主线程阻塞，所有请求卡死 | "CPU密集任务会阻塞Event Loop" |
| **并发控制** | synchronized、Lock | Promise、async/await | "JS没有锁，靠原子操作和队列" |
| **内存模型** | 堆内存共享，栈内存隔离 | 单一堆栈，闭包持有引用 | "闭包可能导致内存泄漏" |
| **错误处理** | try-catch捕获同步异常 | try-catch + .catch()捕获异步 | "未捕获的Promise rejection会崩溃" |
| **典型场景** | 数据库连接池、线程池 | 事件驱动、非阻塞I/O | "Node.js适合I/O密集，不适合CPU密集" |

**面试话术模板**：
> "在Java中，我们用线程池处理并发请求，每个请求独立执行。但在Node.js中，如果我在一个请求里执行了大量计算（比如加密算法），会直接阻塞Event Loop，导致所有用户的请求都卡住。所以我会把CPU密集任务放到Worker Threads或后端Java服务处理。"

### 2. Event Loop 深度解析（必考）

#### 2.1 宏任务 vs 微任务

| 类型 | 包含内容 | 执行时机 | Java类比 |
|------|---------|---------|---------|
| **宏任务** | setTimeout、setInterval、I/O | 每轮Event Loop执行一个 | 类似线程池的任务队列 |
| **微任务** | Promise.then、queueMicrotask | 当前宏任务结束后立即清空 | 类似线程内的优先级队列 |

**执行顺序**：
```
1. 执行同步代码
2. 清空所有微任务队列
3. 执行一个宏任务
4. 重复步骤2-3
```

**经典面试题**：
```javascript
console.log('1');

setTimeout(() => {
  console.log('2');
  Promise.resolve().then(() => console.log('3'));
}, 0);

Promise.resolve().then(() => {
  console.log('4');
  setTimeout(() => console.log('5'), 0);
});

console.log('6');

// 输出顺序：1 6 4 2 3 5
```

**面试话术**：
> "这道题考察的是微任务优先级。第一轮：同步代码1、6；清空微任务4；执行宏任务2，产生微任务3，立即执行；第二轮：执行宏任务5。关键是理解'微任务插队'的机制。"

#### 2.2 Node.js Event Loop 六阶段

| 阶段 | 作用 | 典型API | 面试考点 |
|------|------|---------|---------|
| timers | 执行setTimeout/setInterval | setTimeout | 最小延迟不是0ms |
| pending callbacks | 执行I/O回调 | fs.readFile | 异步I/O的回调在这里 |
| idle, prepare | 内部使用 | - | 可忽略 |
| poll | 等待新的I/O事件 | 网络请求 | 最重要的阶段 |
| check | 执行setImmediate | setImmediate | 比setTimeout优先 |
| close callbacks | 关闭回调 | socket.on('close') | 资源清理 |

**面试必答**：
> "在Next.js的Server Actions中，如果我执行了大量的数据库查询，这些异步I/O会在poll阶段等待。如果某个查询特别慢（比如没加索引），会阻塞整个Event Loop，导致其他用户的请求无法进入。所以我会用数据库连接池 + 超时控制。"

### 3. Promise 并发控制（金融场景必备）

#### 3.1 Promise API对比

| API | 行为 | 适用场景 | 面试考点 |
|-----|------|---------|---------|
| Promise.all | 全部成功才resolve，一个失败就reject | 所有请求都必须成功 | 金融场景慎用 |
| Promise.allSettled | 等待全部完成，返回每个结果 | 对账、批量查询 | **推荐** |
| Promise.race | 第一个完成就返回 | 超时控制 | 配合timeout使用 |
| Promise.any | 第一个成功就返回 | 多数据源降级 | 少用 |

**经典场景：批量对账**
```typescript
// ❌ 错误：一个失败全失败
const results = await Promise.all([
  checkAccount1(),
  checkAccount2(),
  checkAccount3()
]);

// ✅ 正确：全部执行完再处理
const results = await Promise.allSettled([
  checkAccount1(),
  checkAccount2(),
  checkAccount3()
]);

results.forEach((result, index) => {
  if (result.status === 'fulfilled') {
    console.log(`账户${index}对账成功`);
  } else {
    console.error(`账户${index}对账失败:`, result.reason);
  }
});
```

#### 3.2 并发数控制（手写必考）

**场景**：同时调用100个银行接口，但最多只能3个并发，防止打爆后端。

```typescript
async function concurrentLimit<T>(
  tasks: (() => Promise<T>)[],
  limit: number
): Promise<T[]> {
  const results: T[] = [];
  const executing: Promise<void>[] = [];

  for (const [index, task] of tasks.entries()) {
    const promise = task().then(result => {
      results[index] = result;
    });

    executing.push(promise);

    if (executing.length >= limit) {
      await Promise.race(executing);
      executing.splice(
        executing.findIndex(p => p === promise),
        1
      );
    }
  }

  await Promise.all(executing);
  return results;
}

// 使用
const tasks = Array.from({ length: 100 }, (_, i) => 
  () => fetch(`/api/account/${i}`)
);

const results = await concurrentLimit(tasks, 3);
```

**面试话术**：
> "这个并发控制器的核心是用Promise.race等待最快完成的那个，然后从executing数组中移除，给新任务腾位置。这样保证同时最多只有3个请求在跑。在光大银行项目中，我用这个方案批量查询千万级流水，避免了数据库连接池耗尽。"

### 4. 内存管理对比

| 维度 | Java | JavaScript | 面试考点 |
|------|------|-----------|---------|
| **垃圾回收** | 分代回收（Young/Old） | 标记清除 + 增量标记 | "V8的GC策略" |
| **内存泄漏** | ThreadLocal未清理 | 闭包持有引用 | "两者都会泄漏" |
| **排查工具** | jmap、MAT | Chrome DevTools、heapdump | "实战经验" |
| **典型场景** | 大对象进Old区 | 全局变量、定时器 | "如何避免" |

#### 4.1 JavaScript内存泄漏典型场景

**场景1：闭包持有DOM引用**
```javascript
// ❌ 内存泄漏
function createLeak() {
  const largeData = new Array(1000000);
  
  return function() {
    console.log(largeData.length); // 闭包持有largeData
  };
}

const leak = createLeak(); // largeData永远不会被回收

// ✅ 正确
function createNoLeak() {
  const largeData = new Array(1000000);
  const length = largeData.length;
  
  return function() {
    console.log(length); // 只持有基本类型
  };
}
```

**场景2：定时器未清理**
```javascript
// ❌ 内存泄漏
useEffect(() => {
  const timer = setInterval(() => {
    console.log('tick');
  }, 1000);
  // 忘记清理
}, []);

// ✅ 正确
useEffect(() => {
  const timer = setInterval(() => {
    console.log('tick');
  }, 1000);
  
  return () => clearInterval(timer); // 组件卸载时清理
}, []);
```

**面试话术**：
> "在Java中，我遇到过ThreadLocal内存泄漏，需要在finally中调用remove()。在JavaScript中，最常见的是闭包持有大对象引用。比如在React组件中，如果useEffect的依赖数组写错，会导致旧的闭包一直持有引用。我的排查方法是用Chrome DevTools的Heap Snapshot对比前后内存，找到Detached DOM节点。"

### 5. 异步错误处理对比

| 场景 | Java | JavaScript | 最佳实践 |
|------|------|-----------|---------|
| **同步错误** | try-catch | try-catch | 一致 |
| **异步错误** | Future.get() | .catch() / try-catch | 必须捕获 |
| **全局捕获** | Thread.UncaughtExceptionHandler | process.on('unhandledRejection') | 兜底方案 |
| **错误传播** | throws声明 | Promise链式传递 | 不同机制 |

#### 5.1 JavaScript异步错误捕获

**场景1：Promise错误**
```typescript
// ❌ 错误：未捕获的rejection
async function dangerousCode() {
  const data = await fetch('/api/data'); // 如果失败，整个进程可能崩溃
  return data.json();
}

// ✅ 正确：捕获错误
async function safeCode() {
  try {
    const data = await fetch('/api/data');
    return await data.json();
  } catch (error) {
    console.error('请求失败:', error);
    return null; // 返回降级数据
  }
}
```

**场景2：Next.js全局错误处理**
```typescript
// app/error.tsx
'use client';

export default function Error({
  error,
  reset,
}: {
  error: Error & { digest?: string };
  reset: () => void;
}) {
  useEffect(() => {
    // 上报到Sentry
    console.error(error);
  }, [error]);

  return (
    <div>
      <h2>出错了</h2>
      <button onClick={reset}>重试</button>
    </div>
  );
}
```

**面试话术**：
> "在Java中，我们用try-catch捕获同步异常，用Future.get()捕获异步异常。在JavaScript中，async/await让异步代码看起来像同步，但本质还是Promise。如果不加try-catch，未捕获的rejection会触发unhandledRejection事件，在Node.js中可能导致进程退出。所以我在Next.js项目中，会用error.tsx统一捕获，并上报到Sentry。"
